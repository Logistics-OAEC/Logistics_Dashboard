// Tab Navigation function switchTab(evt, tabId) { document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active')); document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active')); document.getElementById(tabId).classList.add('active'); evt.currentTarget.classList.add('active'); if (tabId === 'report-tab') { generateReport(); } } // Extract table data dynamically from existing DOM table function getLogData() { const rows = document.querySelectorAll('.table tbody tr'); // Adjust selector if needed const data = []; rows.forEach(row => { const cols = row.querySelectorAll('td'); if (cols.length >= 8) { data.push({ type: cols[0].innerText.trim(), mobType: cols[1].innerText.trim(), cargo: cols[2].innerText.trim(), date: cols[3].innerText.trim(), hauler: cols[4].innerText.trim(), source: cols[5].innerText.trim(), destination: cols[6].innerText.trim(), status: cols[7].innerText.trim() }); } }); return data; } // Generate Summary Report function generateReport() { const requests = getLogData(); document.getElementById('report-timestamp').innerText = new Date().toLocaleString(); const total = requests.length; const pending = requests.filter(r => r.status.toLowerCase().includes('pending')).length; const served = requests.filter(r => r.status.toLowerCase().includes('served')).length; const rate = total > 0 ? ((served / total) * 100).toFixed(1) : 0; document.getElementById('rep-total').innerText = total; document.getElementById('rep-pending').innerText = pending; document.getElementById('rep-served').innerText = served; document.getElementById('rep-rate').innerText = `${rate}%`; // Aggregate stats const typeStats = {}; const haulerStats = {}; requests.forEach(r => { // Type Breakdown if (!typeStats[r.type]) typeStats[r.type] = { total: 0, pending: 0, served: 0 }; typeStats[r.type].total++; if (r.status.toLowerCase().includes('served')) typeStats[r.type].served++; else typeStats[r.type].pending++; // Hauler Breakdown if (!haulerStats[r.hauler]) haulerStats[r.hauler] = { total: 0, pending: 0, served: 0 }; haulerStats[r.hauler].total++; if (r.status.toLowerCase().includes('served')) haulerStats[r.hauler].served++; else haulerStats[r.hauler].pending++; }); // Render Type Table const typeTbody = document.querySelector('#type-breakdown-table tbody'); typeTbody.innerHTML = Object.entries(typeStats).map(([type, stats]) => ` ${type} ${stats.total} ${stats.pending} ${stats.served} `).join('') || 'No data available'; // Render Hauler Table const haulerTbody = document.querySelector('#hauler-breakdown-table tbody'); haulerTbody.innerHTML = Object.entries(haulerStats).map(([hauler, stats]) => ` ${hauler} ${stats.total} ${stats.served} ${stats.pending} `).join('') || 'No data available'; } // Export CSV function exportCSV() { const requests = getLogData(); let csv = 'Type,Mobilization Type,Cargo,Date Requested,Unit Hauler,Source,Destination,Status\n'; requests.forEach(r => { csv += `"${r.type}","${r.mobType}","${r.cargo}","${r.date}","${r.hauler}","${r.source}","${r.destination}","${r.status}"\n`; }); const blob = new Blob([csv], { type: 'text/csv' }); const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = `Logistics_Report_${new Date().toISOString().slice(0, 10)}.csv`; a.click(); }
